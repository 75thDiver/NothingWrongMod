# All effects enter in the receiving country scope.
nw_nr_first_mobilization = {
	if = {
		limit = {
			OR = { tag = SOV tag = UKR tag = DNR }
			NOT = { has_global_flag = nw14_russian_cw_ended }
			NOT = { has_country_flag = nw_nr_first_mobilization_granted }
		}
		set_country_flag = nw_nr_first_mobilization_granted
		if = {
			limit = { tag = SOV }
			set_variable = { nw_nr_reinforcement_count = 36 }
		}
		else_if = {
			limit = { tag = UKR }
			set_variable = { nw_nr_reinforcement_count = 6 }
		}
		else = { set_variable = { nw_nr_reinforcement_count = 3 } }
		nw_nr_spawn_militia = yes
		clear_variable = nw_nr_reinforcement_count
	}
}

nw_nr_reinforce_middle = {
	if = {
		limit = {
			NOT = { has_global_flag = nw14_russian_cw_ended }
			NOT = { has_country_flag = nw_nr_reinforcements_middle_granted }
		}
		set_country_flag = nw_nr_reinforcements_middle_granted
		set_variable = { nw_nr_reinforcement_count = 1 }
		if = {
			limit = { tag = SOV }
			set_variable = { nw_nr_reinforcement_count = 5 }
			nw_nr_spawn_militia = yes
			nw_nr_spawn_regular_brigades = yes
		}
		else_if = {
			limit = { tag = DNR }
			nw_nr_spawn_don_cavalry = yes
		}
		else = {
			if = {
				limit = { OR = { tag = BLR tag = FER tag = UKR } }
				set_variable = { nw_nr_reinforcement_count = 3 }
			}
			nw_nr_spawn_light_brigades = yes
		}
		clear_variable = nw_nr_reinforcement_count
		custom_effect_tooltip = nw_nr_front_bonus_5_tt
		hidden_effect = { nw_nr_apply_front_bonus = yes }
	}
}

# Reconcile combat rewards without running the division grants again.
nw_nr_apply_front_bonus = {
	if = {
		limit = {
			OR = {
				has_country_flag = nw_nr_reinforcements_late_granted
				check_variable = { nw_nr_revolutionary_attack > 0.075 }
			}
		}
		remove_ideas = nw_nr_revolutionary_front_1
		add_ideas = nw_nr_revolutionary_front_2
	}
	else_if = {
		limit = {
			OR = {
				has_country_flag = nw_nr_reinforcements_middle_granted
				check_variable = { nw_nr_revolutionary_attack > 0 }
			}
		}
		add_ideas = nw_nr_revolutionary_front_1
	}
	if = {
		limit = { check_variable = { nw_nr_revolutionary_attack > 0 } }
		# The old branch added the same amount of generic attack and defense.
		# Leave the defense from United Front and the congress event in place.
		subtract_from_variable = { var = nw_nr_defence value = nw_nr_revolutionary_attack }
		clear_variable = nw_nr_revolutionary_attack
	}
	force_update_dynamic_modifier = yes
}

# The engine can move a fraction of the whole army but cannot transfer selected
# divisions. Re-form the allied quota in friendly capitals and refund only
# the demobilized cohort. Other SOV templates are untouched.
nw_nr_reorganize_sov_militia = {
	hidden_effect = {
		set_temp_variable = { nw_nr_militia_total = 0 }
		every_country_division = {
			limit = { division_has_majority_template = militia }
			add_to_temp_variable = { nw_nr_militia_total = 1 }
		}
		if = {
			limit = { check_variable = { nw_nr_militia_total > 0 } }
			set_temp_variable = { nw_nr_militia_recipients = 0 }
			every_country = {
				limit = { nw_nr_can_receive_sov_militia = yes }
				add_to_temp_variable = { nw_nr_militia_recipients = 1 }
			}
			set_temp_variable = { nw_nr_militia_to_transfer = 0 }
			if = {
				limit = { check_variable = { nw_nr_militia_recipients > 0 } }
				set_temp_variable = { nw_nr_militia_to_transfer = nw_nr_militia_total }
				divide_temp_variable = { nw_nr_militia_to_transfer = 2 }
				subtract_from_temp_variable = { nw_nr_militia_to_transfer = 0.25 }
				round_temp_variable = nw_nr_militia_to_transfer
			}
			set_temp_variable = { nw_nr_militia_to_mark = nw_nr_militia_to_transfer }
			division_template = {
				name = "NW14 Militia Handover"
				priority = 1
				regiments = {
					militia = { x = 0 y = 0 }
					militia = { x = 0 y = 1 }
					militia = { x = 1 y = 0 }
					militia = { x = 1 y = 1 }
					militia = { x = 2 y = 0 }
					militia = { x = 2 y = 1 }
				}
			}
			division_template = {
				name = "NW14 Militia Demobilization"
				priority = 1
				regiments = {
					militia = { x = 0 y = 0 }
					militia = { x = 0 y = 1 }
					militia = { x = 1 y = 0 }
					militia = { x = 1 y = 1 }
					militia = { x = 2 y = 0 }
					militia = { x = 2 y = 1 }
				}
			}
			every_country_division = {
				limit = { division_has_majority_template = militia }
				if = {
					limit = { check_variable = { nw_nr_militia_to_mark > 0 } }
					change_division_template = { division_template = "NW14 Militia Handover" }
					subtract_from_temp_variable = { nw_nr_militia_to_mark = 1 }
				}
				else = {
					change_division_template = { division_template = "NW14 Militia Demobilization" }
				}
			}
			delete_unit_template_and_units = {
				division_template = "NW14 Militia Demobilization"
				disband = yes
			}
			delete_unit_template_and_units = {
				division_template = "NW14 Militia Handover"
				disband = no
			}
			if = {
				limit = { check_variable = { nw_nr_militia_to_transfer > 0 } }
				every_country = {
					limit = { nw_nr_can_receive_sov_militia = yes }
					# Allocate the nearest integer share of the remaining quota.
					set_temp_variable = { nw_nr_allied_share = nw_nr_militia_to_transfer }
					divide_temp_variable = { nw_nr_allied_share = nw_nr_militia_recipients }
					round_temp_variable = nw_nr_allied_share
					if = {
						limit = { check_variable = { nw_nr_allied_share > 0 } }
						set_variable = { nw_nr_reinforcement_count = nw_nr_allied_share }
						nw_nr_spawn_militia = yes
						clear_variable = nw_nr_reinforcement_count
					}
					subtract_from_temp_variable = { nw_nr_militia_to_transfer = nw_nr_allied_share }
					subtract_from_temp_variable = { nw_nr_militia_recipients = 1 }
				}
			}
		}
	}
}
