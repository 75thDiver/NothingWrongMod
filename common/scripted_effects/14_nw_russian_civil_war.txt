# Country scope. Each iterated country is an actual AFS enemy; inside a
# subject scope PREV is that enemy. Syntax follows vanilla events/Japan.txt.
nw14_rus_civil_war_join_white_subjects = {
	every_country = {
		limit = {
			has_war_with = AFS
			OR = { tag = SOV tag = RUK tag = FER tag = ABK tag = NCR tag = TCF tag = BLW tag = BLR tag = KIR tag = DNR tag = KAL }
		}
		DON = {
			if = {
				limit = { exists = yes is_subject_of = AFS NOT = { has_war_with = PREV } }
				add_to_war = { targeted_alliance = AFS enemy = PREV hostility_reason = asked_to_join }
			}
		}
		KUB = {
			if = {
				limit = { exists = yes is_subject_of = AFS NOT = { has_war_with = PREV } }
				add_to_war = { targeted_alliance = AFS enemy = PREV hostility_reason = asked_to_join }
			}
		}
		AKV = {
			if = {
				limit = { exists = yes is_subject_of = DON NOT = { has_war_with = PREV } }
				add_to_war = { targeted_alliance = DON enemy = PREV hostility_reason = asked_to_join }
			}
		}
	}
}
