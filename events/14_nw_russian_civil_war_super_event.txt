add_namespace = nw14cw

# Presentation follows Kaiserredux's hidden event -> scripted GUI -> sound effect.
country_event = {
	id = nw14cw.1
	hidden = yes
	is_triggered_only = yes
	fire_only_once = yes

	immediate = {
		if = {
			limit = { NOT = { has_global_flag = nw14_cw_super_event_announced } }
			set_global_flag = nw14_cw_super_event_announced
			every_country = {
				limit = { is_ai = no }
				set_country_flag = nw14_cw_super_event_visible
				scoped_sound_effect = nw14_cw_super_event_sound
			}
		}
	}
}

# Poland's fall prompts an independent Finland to choose its side in the war.
country_event = {
	id = nw14cw.4
	title = nw14cw.4.t
	desc = nw14cw.4.d
	fire_only_once = yes
	is_triggered_only = yes
	trigger = {
		tag = FIN
		has_country_flag = nw14_fin_white_offer_fired
		has_country_flag = nw14_cw_active
		has_capitulated = no
		is_subject = no
		event_target:nw14_rus_royal = { exists = yes }
	}
	option = {
		name = nw14cw.4.a
		ai_chance = { factor = 75 }
		# Target only the White Movement and its established subject chain.
		# No white_peace is explicitly issued against New Russia.
		if = {
			limit = { event_target:nw14_rus_royal = { exists = yes has_war_with = ROOT } }
			event_target:nw14_rus_royal = { white_peace = { tag = ROOT } }
		}
		if = {
			limit = { AFS = { exists = yes has_war_with = ROOT } }
			AFS = { white_peace = { tag = ROOT } }
		}
		if = {
			limit = { DON = { exists = yes has_war_with = ROOT } }
			DON = { white_peace = { tag = ROOT } }
		}
		if = {
			limit = { KUB = { exists = yes has_war_with = ROOT } }
			KUB = { white_peace = { tag = ROOT } }
		}
		if = {
			limit = { AKV = { exists = yes has_war_with = ROOT } }
			AKV = { white_peace = { tag = ROOT } }
		}
		if = { limit = { is_in_faction = yes } leave_faction = yes }
		event_target:nw14_rus_royal = { FIN = { add_to_faction = PREV } }
	}
	option = {
		name = nw14cw.4.b
		ai_chance = { factor = 25 }
	}
}
