# ROOT is the dominion. SOV is deliberately explicit for the fixed Russian route.

country_event = {
	id = nw24.3
	title = nw24.3.t
	desc = nw24.3.d
	is_triggered_only = yes

	option = {
		name = nw24.3.a
		trigger = {
			is_subject_of = SOV
		}
		ai_chance = {
			factor = 70
		}
		clr_country_flag = nw24_charter_pending
		set_country_flag = nw24_charter_signed
		add_stability = 0.03
		add_political_power = 25
		SOV = {
			add_to_variable = {
				var = nw_trust
				value = 5
			}
			clamp_variable = {
				var = nw_trust
				min = 0
				max = 100
			}
			nw24_refresh_compact = yes
			country_event = {
				id = nw24.4
				days = 1
			}
		}
	}

	option = {
		name = nw24.3.b
		trigger = {
			is_subject_of = SOV
		}
		ai_chance = {
			factor = 30
		}
		clr_country_flag = nw24_charter_pending
		SOV = {
			add_to_variable = {
				var = nw_trust
				value = -5
			}
			clamp_variable = {
				var = nw_trust
				min = 0
				max = 100
			}
			country_event = {
				id = nw24.4
				days = 1
			}
		}
	}

	option = {
		name = nw24.3.c
		trigger = {
			NOT = {
				is_subject_of = SOV
			}
		}
		clr_country_flag = nw24_charter_pending
	}
}
