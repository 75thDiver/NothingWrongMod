	title = nw26.1.t
	desc = nw26.1.d
	is_triggered_only = yes
	trigger = {
		has_country_flag = nw26_admin_pending
		NOT = {
			has_country_flag = nw26_admin_settled
		}
	}

	option = {
		name = nw26.1.a
		trigger = {
			has_country_flag = nw26_admin_pending
			NOT = {
				has_country_flag = nw26_admin_settled
			}
		}
		ai_chance = {
			factor = 50
		}
		if = {
			limit = {
				has_country_flag = nw26_admin_pending

				}
			}
			clr_country_flag = nw26_admin_pending
			set_country_flag = nw26_admin_settled
			add_ideas = nw26_gen_administration
		}
	}

	option = {

			set_country_flag = nw26_admin_settled

	option = {
		name = nw26.1.c
